Integrating a shader into the homepage 
// 1. Copy a shader with "Copy GLSL" (it includes the shared prelude).
// 2. Draw the day's portrait into any canvas/img, then:
const gl = canvas.getContext("webgl");
const prog = link(gl, VERT, fragmentSourceFromGallery);
// uniforms: uTex (portrait), uRes (px), uTime (s), uP (0..1 intensity)
function frame(t) {
  gl.uniform1f(uTime, t / 1000);
  gl.uniform1f(uP, 0.5);
  gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
  requestAnimationFrame(frame);
}
// Tip: pick shader = dayIndex % 14, so each of the 180 days has a different "decrunch".
